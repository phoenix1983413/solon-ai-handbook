17.1 理论讲解

17.1.1 IoT 监控的核心挑战

行业痛点： 1. 设备规模大：万级设备并发上报，数据量巨大 2. 实时性要求高：告警延迟需要秒级甚至毫秒级 3. 协议碎片化：MQTT/CoAP/Modbus/HTTP 多种协议 4. 故障诊断难：设备分散，远程调试困难 5. 数据存储成本高：时序数据长期存储成本高

AI 赋能 IoT 监控的三个层次： 1. 辅助层：智能告警（减少误报）、故障预测 2. 智能层：根因分析、自动诊断 3. 决策层：预防性维护、设备健康评分

17.1.2 核心技术

1. 时序数据处理

  • 时间序列数据库（InfluxDB/TDEngine/Prometheus）
  • 降采样（Downsampling）策略
  • 数据保留策略（热数据/温数据/冷数据）

2. 异常检测

  • 统计方法（3-sigma、Z-score）
  • 机器学习方法（Isolation Forest、LSTM-Autoencoder）
  • LLM 辅助分析（结合领域知识）

3. 故障预测

  • 基于时序预测（Prophet、ARIMA）
  • 基于设备健康指标（振动、温度、功耗）
  • 基于历史故障模式

4. 远程诊断

  • MCP 协议调用设备 API
  • Agent 自动执行诊断流程
  • 生成诊断报告

17.1.4 关键技术决策

Q1：时序数据库选型

数据库 适用场景 优点 缺点
InfluxDB 中小规模 易用、生态好 集群版收费
TDEngine 大规模时序 高性能、开源 学习成本
Prometheus 监控指标 生态完善 不适合设备数据
TimescaleDB PostgreSQL 扩展 兼容 SQL 依赖 PG

推荐：TDEngine（开源、高性能、适合设备规模大场景）

Q2：异常检测算法选择

算法 适用场景 优点 缺点
3-sigma 单指标、正态分布 简单快速 无法处理多指标
Isolation Forest 多指标、异常稀少 无需标注 可解释性差
LSTM-Autoencoder 时序模式 捕捉时序依赖 训练成本高
LLM 辅助 复杂故障 结合领域知识 成本高

推荐：3-sigma 做实时过滤，Isolation Forest 做批量检测，LLM 做根因分析

Q3：告警降噪策略

策略 说明
时间窗口 5 分钟内相同告警合并
依赖关系 上游告警触发下游告警抑制
历史基线 与历史同期对比，排除周期性波动
LLM 分析 结合多指标判断是否为真实故障

    }

练习2：基于规则的实时异常检测

@Singleton
public class RuleBasedAnomalyDetector {
    private final AlertService alertService;

    public void checkAnomaly(Device device, TelemetryData data) {
        List<Anomaly> anomalies = new ArrayList<>();

        // 1. 温度异常（> 80°C）
        if (data.getTemperature() > 80) {
            anomalies.add(new Anomaly("TEMPERATURE_HIGH", 
                String.format("温度过高: %.1f°C", data.getTemperature())));
        }

        // 2. 电压异常（< 3.0V 或 > 4.2V）
        if (data.getVoltage() < 3.0 || data.getVoltage() > 4.2) {
            anomalies.add(new Anomaly("VOLTAGE_OUT_OF_RANGE",
                String.format("电压异常: %.2fV", data.getVoltage())));
        }

        // 3. 电流异常（> 5A）
        if (data.getCurrent() > 5.0) {
            anomalies.add(new Anomaly("CURRENT_HIGH",
                String.format("电流过高: %.2fA", data.getCurrent())));
        }

        // 4. 发送告警
        for (Anomaly anomaly : anomalies) {
            alertService.send(device.getId(), anomaly);
        }
    }
}

练习3：时序数据异常检测（3-sigma）

@Singleton
public class TimeSeriesAnomalyDetector {
    private final TdEngineClient tdEngineClient;

    public boolean isAnomaly(String deviceId, String metric, long timestamp) {
        // 1. 获取历史数据（最近 24 小时）
        List<Double> history = tdEngineClient.query(deviceId, metric, 
            timestamp - 24 * 3600 * 1000, timestamp);

        if (history.size() < 10) {
            return false; // 数据不足
        }

        // 2. 计算均值和标准差
        double mean = history.stream().mapToDouble(d -> d).average().orElse(0);
        double stdDev = Math.sqrt(history.stream()
                .mapToDouble(d -> Math.pow(d - mean, 2))
                .average().orElse(0));

        // 3. 获取最新值
        double latest = history.get(history.size() - 1);

        // 4. 3-sigma 判断
        return Math.abs(latest - mean) > 3 * stdDev;
    }
}

@Singleton
public class DeviceIngestionService {
    private final TdEngineClient tdEngineClient;
    private final DeviceRepository deviceRepository;
    private final AnomalyDetector anomalyDetector;

    @MqttSubscribe("device/+/telemetry")
    public void onTelemetry(String topic, TelemetryData data) {
        String deviceId = extractDeviceId(topic);

        // 1. 验证设备
        Device device = deviceRepository.findById(deviceId)
                .orElseThrow(() -> new DeviceNotFoundException(deviceId));

        // 2. 存储时序数据
        tdEngineClient.insert("telemetry", new TelemetryRecord(deviceId, data));

        // 3. 实时异常检测
        List<Anomaly> anomalies = anomalyDetector.detect(device, data);
        if (!anomalies.isEmpty()) {
            alertService.send(device, anomalies);
        }

        // 4. 更新设备状态
        device.setLastSeen(System.currentTimeMillis());
        deviceRepository.update(device);
    }
}

模块2：时序数据查询服务

@RestController
@RequestMapping("/api/telemetry")
public class TelemetryController {
    private final TdEngineClient tdEngineClient;

    @Get("/{deviceId}")
    public TimeSeriesData query(
            @Path String deviceId,
            @Param("metric") String metric,
            @Param("start") long start,
            @Param("end") long end,
            @Param("interval") String interval) {

        // 1. 查询原始数据
        List<TelemetryPoint> points = tdEngineClient.query(
            deviceId, metric, start, end, interval
        );

        // 2. 降采样（如果查询范围过大）
        if (points.size() > 1000) {
            points = downsample(points, 1000);
        }

        return TimeSeriesData.builder()
                .deviceId(deviceId)
                .metric(metric)
                .points(points)
                .build();
    }
}

模块4：智能告警引擎

@Singleton
public class SmartAlertEngine {
    private final AlertRepository alertRepository;
    private final NotificationService notificationService;

    public void process(Anomaly anomaly) {
        // 1. 去重（相同设备、相同类型、5分钟内）
        Alert recentAlert = alertRepository.findRecent(anomaly.getDeviceId(), 
            anomaly.getType(), System.currentTimeMillis() - 5 * 60 * 1000);
        if (recentAlert != null) {
            return; // 抑制重复告警
        }

        // 2. 关联分析（检查依赖设备）
        boolean isCascade = checkCascadeFailure(anomaly);

        // 3. 生成告警
        Alert alert = Alert.builder()
                .deviceId(anomaly.getDeviceId())
                .type(anomaly.getType())
                .severity(isCascade ? AlertSeverity.CRITICAL : AlertSeverity.WARNING)
                .message(anomaly.getMessage())
                .timestamp(System.currentTimeMillis())
                .build();

        alertRepository.save(alert);

        // 4. 发送通知
        notificationService.send(alert);
    }

    private boolean checkCascadeFailure(Anomaly anomaly) {
        // 检查是否因上游设备故障导致
        Device device = deviceRepository.findById(anomaly.getDeviceId()).orElseThrow();
        if (device.getDependsOn() != null) {
            Alert upstreamAlert = alertRepository.findRecent(device.getDependsOn(),
                AlertType.OFFLINE, System.currentTimeMillis() - 10 * 60 * 1000);
            return upstreamAlert != null;
        }
        return false;
    }
}

模块5：监控大屏

@RestController
@RequestMapping("/api/dashboard")
public class DashboardController {
    private final TdEngineClient tdEngineClient;
    private final DeviceRepository deviceRepository;

    @Get("/overview")
    public DashboardOverview getOverview() {
        // 1. 设备总数
        long totalDevices = deviceRepository.count();

        // 2. 在线设备数
        long onlineDevices = deviceRepository.countOnline();

        // 3. 今日告警数
        long todayAlerts = alertRepository.countToday();

        // 4. 关键指标趋势
        List<TimeSeriesPoint> cpuTrend = tdEngineClient.queryAggregate(
            "cpu_usage", "1h", "AVG"
        );

        return DashboardOverview.builder()
                .totalDevices(totalDevices)
                .onlineDevices(onlineDevices)
                .todayAlerts(todayAlerts)
                .cpuTrend(cpuTrend)
                .build();
    }
}
